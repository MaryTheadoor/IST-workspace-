% Pandoc + XeLaTeX preamble for IST (v7.0)
% Maps literal Unicode Greek / operators / superscripts found in the
% markdown (captions, tables, headings) to LaTeX glyphs, since the text
% font (Latin Modern) lacks them outside math mode.

\usepackage{float}

% Compact figures: scale down to fit better on letter paper
\setkeys{Gin}{width=0.85\linewidth,height=0.45\textheight,keepaspectratio}

\newunicodechar{σ}{\ensuremath{\sigma}}

\newunicodechar{θ}{\ensuremath{\theta}}
