class Thread:
    def __init__(self, elements=None):
        self.elements = elements or []
        self.time = 0
        self.twisted = False

    def push(self, dn: DirectedNumber):
        self.elements.append(dn)

    def pop(self) -> DirectedNumber:
        return self.elements.pop()

    def fork(self):
        new_thread = Thread(self.elements.copy())
        return new_thread

    def join(self, other: 'Thread'):
        # merge by element-wise multiplication
        min_len = min(len(self.elements), len(other.elements))
        new_elems = [mul(self.elements[i], other.elements[i]) for i in range(min_len)]
